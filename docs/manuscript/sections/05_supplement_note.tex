% ── S1: Vital signs from M4 ─────────────────────────────────────────────────
\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.85\linewidth]{figS02_vitals_m4.png}
  \caption{\textbf{Vital-sign hazard ratios from the M4 model.}
  Initial heart rate, respiratory rate, and systolic blood pressure (each z-scored) are adjusted for in M3 and M4 but are not the focus of primary inference.
  All three vital signs show associations in the expected direction: higher heart rate and respiratory rate are associated with faster reassessment, consistent with physiologic acuity driving monitoring intensity.}
  \label{fig:vitals_supp}
\end{figure}

% ── S2: Sequential attenuation — workflow variables ─────────────────────────
\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.9\linewidth]{figS03_attenuation_workflow.png}
  \caption{\textbf{Sequential attenuation for ED workflow variables across M1--M4.}
  Arrival mode, shift, and crowding variables are entered at the M4 tier.
  The M4 panel shows final adjusted estimates; panels M1--M3 show partially adjusted estimates from models not yet including these variables.}
  \label{fig:attenuation_workflow}

% ── S3: Disposition-stratified M4 forests ───────────────────────────────────
\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.95\linewidth]{figS04_disposition_stratified.png}
  \caption{\textbf{Disposition-stratified M4 forest plots.}
  M4 models fit separately among patients discharged home and patients admitted to the hospital.
  Disposition is not included as a covariate in primary models because it is downstream of initial pain documentation; these stratified models assess whether the primary associations are consistent across disposition groups.}
  \label{fig:disposition_supp}
\end{figure}

% ── S4: Within-acuity heatmaps ───────────────────────────────────────────────
\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{figS05_within_acuity_heatmap.png}
  \caption{\textbf{Within-acuity heatmaps for additional covariates.}
  Estimated hazard ratios from M4-style models within ESI strata (1--2, 3, 4--5) for age group, sex, diagnosis, and preferred language.
  Color encodes the HR direction and magnitude; cells with wide confidence intervals are shaded accordingly.
  These analyses complement the main within-acuity results for insurance and race/ethnicity (Figure~\ref{fig:withinacuity} in main text).}
  \label{fig:acuity_heatmap}
\end{figure}

% ── S5: IPTW balance and weighted Cox ────────────────────────────────────────
\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.9\linewidth]{figS06_iptw_balance.png}
  \caption{\textbf{IPTW covariate balance and weighted Cox estimate for Medicaid vs.\ private insurance.}
  \textit{Left:} Standardized mean differences before and after inverse probability of treatment weighting (IPTW), using propensity scores estimated from M1--M3 covariates.
  \textit{Right:} Weighted Cox hazard ratio for Medicaid vs.\ private insurance, compared with the unweighted M4 estimate.
  The IPTW estimate corroborates the primary finding of slower reassessment among Medicaid patients and is not meaningfully attenuated relative to the regression-based estimate.}
  \label{fig:iptw_supp}
\end{figure}

% ── S6: Continuous-year Cox model ────────────────────────────────────────────
\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.75\linewidth]{figS07_continuous_year_cox.png}
  \caption{\textbf{Year-over-year Cox model (M4 covariates).}
  Smoothed hazard ratio for reassessment timing as a function of de-identified year, estimated from an M4 model including a restricted cubic spline for calendar time.
  The flat trajectory confirms the absence of a meaningful change in reassessment rates over the observation window (HR per 5-year increase \HRYearFive{}).}
  \label{fig:year_supp}
\end{figure}

% ── S7: IPTW weighted KM ─────────────────────────────────────────────────────
\begin{figure}[htbp]

  Weighted cumulative incidence of reassessment after IPTW balancing on M1--M3 covariates.
  The persistent separation between Medicaid and private insurance in the weighted curves is consistent with the primary regression estimate and the within-acuity analyses.}
